import std.io

fn test_arrays()
{
    // type will be infered from first element
    var x = [1, 2, 3]
    // size infered
    var y: i32[] = [4, 5, 6]
    // explicit size
    var z: i32[3] = [7, 8, 9]

    // array programming
    var result = x + y + z

    println(result)

    // ranged based loop
    for n in result
    {
        println(n)
    }

    import std.containers

    // can also do List<str> but will infer
    var names = List{"John"}

    names.add("henry")

    // all containers are similar types forbid copying and must either clone or move explicitly or take a reference
    var names_copy = ^names

    // @ moves the object and will reset it to its initial state. length = 0 capacity = 0 and data = nil
    // moved values are safe to use as long as their initial state is also safe to use
    var names_moved = @names

    // now refers to an empty names list
    // when add is called again it will reallocate the memory
    var names_ptr = &names

    // will use an allocator that can support 256 hash table entries on the stack
    var numbers = Map<string, int, StackAllocator<256>>{}
}

// any object can be iterable if it has a length field and implements op_index 
// the compiler will then turn it into for var i in 0..list.length { var item = list[i] }
// allocator will default to whatever is the scopes allocator at the time of creation unless specified otherwise
@Readonly
struct CustomList<T, A = #allocator<T>> : Drop, Copy
{
    length: int
    capacity: int
    allocator: A
    ptr: *T
}

// no need to redefine generics
impl CustomList 
{
    // accepts any number of T parameters
    Self(T... args)
    {
        s.ensure_size(args.length)

        $for item in args 
        {
            s.add(item)
        }
    }

    // called automatically when no longer in scope
    fn drop()
    {
        if s.ptr != nil 
        {
            s.allocator.free(s.ptr)
        }

        s.length = 0
        s.capacity = 0
        s.ptr = nil
    }

    fn op_index(index: int) => s.ptr[index]

    // will be called when slicing 
    fn op_slice(start, end: int) => s.ptr[start:end]

    fn copy(): Self
    {
        // $bit_copy is allowed even when copying is disabled
        // self and s are always equivalent but self should be preferred as a value
        // and s for member access
        var new_self = $bit_copy(self)

        new_self.ptr = new_self.allocator.alloc(new_self.capacity)

        import std.mem

        memcpy(s.ptr, new_self.ptr)

        return new_self
    }

    fn ensure_size(length: int = 1)
    {
        // length will shadow self.length and self.length has to be used explicitly
        if s.length + length >= capacity
        {
            s.capacity *= 2
            allocator.realloc(s.ptr, s.capacity)
        }
    }

    fn add(T item): *T
    {
        s.ensure_size()

        var index = s.length++

        s.ptr[index] = item

        return &s.ptr[index]
    }
}

fn custom_list_test()
{
    // will infer from first argument
    // the rest can then be coerced into that type if possible
    var list = CustomList{1, 2, 3}

    // will do a deep copy of list
    var list2 = ^list

    // list2 will now be length 0 capacity 0 and ptr nil
    // on next call to add it will reallocate and be back to a valid state
    var list3 = @list2

    list3.add(10)

    // explicitly drop resource
    drop list3

    // default initializer works fine
    var list4 = CustomList<int>{}
}
