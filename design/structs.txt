struct Entity
{
    id: int
}

struct MyStruct
{
    // now all fields of Entity will also be a part of my struct.
    // the fields are added where ever the mixin is used
    // mixins can also be used within an impl block to mixin methods
    mixin Entity

    x: int
    y: float
    s: str
}

impl MyStruct 
{
    // compiler will now call this when doing ~my_struct
    fn op_stringify(s) => $"{s.x} {s.y} {s.s}"
}

fn print_struct<T>(value: &&T)
{
    for field in $fields(value)
    {
        println($"{field.name}: {field.type} = {field.value}")
    }
}

trait Shape
{
    fn get_area(): float
}

struct Square
{
    width: float
    height: float
}

impl Square : Shape
{
    fn get_area() => width * height
}

// the compiler will use duck typing instead where possible.
// when calling print_area because this has a matching function signature it will also consider this to be ok
// traits have the advantage of working at runtime as well as compile time
struct FakeShape
{
    fn get_area() => 0.0f
}

fn make_shape() => Square {width: 1024, height: 1920}

fn print_area(shape: Shape) => println(shape.get_area())

fn test_shape()
{
    var shape = make_shape()

    print_shape(shape)
}

fn test_dyn()
{
    // scope_alloc will add a defer #free call automatically
    let shape = #scope_alloc(Square{width: 1024, height: 1920})

    var fake_shape: ^Shape

    // this is a compiler error. it does not have the trait so it only works for compile time duck typing
    fake_shape = #alloc<FakeShape>()
}

struct TypeWrapper<T>
{
    @alias_this
    value: T
}

fn test_wrapper()
{
    var wrapper = TypeWrapper{100}

    wrapper++

    println(wrapper) // should print 101
}

struct Data
{
    value: i32
}

impl Data
{
    Self() => {100}
}

fn test_default()
{
    // here the compiler will insert a call to Data::Self and initialize itself to 100
    var data: Data

    data.value = 10

    // again Data::Self is called and the value is set back to 100
    data = {}

    // because of the undefined keyword Data::Self is not called and no default value will be set.
    var data2: Data = undefined
}

// class types are private by default
// methods are still public by default
class SecretData
{
    password: str
}

// fields are public by default
struct Player
{
    data: PlayerData
    private secret_data: bool
}

impl Player
{
    fn print_stats() => println($"{self.name}: {self.score}")

    @static
    fn static_function() => println("this is a static call")

    // function overloading that only allows + and -
    fn op_compare<Op: str>(rhs: &Self) when Op == ("+", "-") => mixin($"{self.score} {Op} {rhs.score}")
}
