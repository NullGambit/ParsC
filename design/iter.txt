
struct Data<T, N>
{
    array: [N]T
}

struct DataIterator<T, N> 
{
    data: Data<T, N>
    cursor: u32
}

impl DataIterator : Iterator 
{
    fn has_next() => self.cursor < N 
    fn get_next() => (self.data.array[self.cursor], self.cursor ** 2)
}

impl Data : Iterable
{
    fn iter() => DataIterator{self, 0}
}
