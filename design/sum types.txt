alias JsonValue = i32 | f32 | str | bool

fn test_unions(value: JsonValue)
{
    match value
    {
        i32 => println("value is an int"),
        _ => println("has some other value")
    }

    if value is string
    {
        // value will only be a string here
        println(value)
    }
}

// a union unlike a sum type is identical to a C union except it can have methods like a regular 
// struct and each member can be named 
// useful sometimes
union JsonUnion 
{
    i: i32
    f: f32 
    s: str 
    b: bool 
}

enum JsonKind 
{
    Int
    Float
    String
    Boo
}

// tagged union
struct JsonValue 
{
    as: JsonUnion 
    kind: JsonKind
}

// rust style enum payload
enum JsonValue 
{
    Int(i32)
    Float(f32)
    String(str)
    Bool(bool)
}

// may seem redundant to have so many different forms of algebraic types, even the error unions are another form of algebraic types 
// but each form is useful for some specific reason and i prefer to have all to more naturally represent different class of problems.
